\section{Motivation}

\begin{frame}{Machines Now Write Code. People Still Have to Accept It.}
\begin{itemize}
  \item AI coding agents open pull requests directly: they author changes,
        but they do not evaluate, explain, revise, or accept them.
  \item Evaluation stays human. A technology that changes the volume and
        character of proposed changes alters the review task
        \emph{without performing it}.
  \item The organizational question is not whether coding gets faster.
        It is how the complementary evaluation work is redistributed
        among the people still responsible for it.
\end{itemize}
\vfill
\begin{block}{Two competing expectations in practice}
  Machine output is waved through (rubber-stamping), \emph{or} it piles
  extra load onto a few maintainers (concentration). We test both.
\end{block}
\end{frame}

\section{Theory}

\begin{frame}{Theory Makes a Sharp Prediction: Concentration}
\begin{itemize}
  \item In a knowledge-based hierarchy, matching a problem to someone who
        can solve it is costly. Those nearer production handle the common
        and easy problems; a smaller set of specialists absorbs the rare
        and hard ones.
  \item A technology absorbing \emph{routine} production removes the easy
        cases disproportionately.
  \item So the residual evaluation work is weighted toward exceptions, and
        exceptions route upward to the most knowledgeable reviewers.
\end{itemize}
\vfill
\begin{alertblock}{Prediction}
  Review should \textbf{concentrate} on a project's busiest reviewers
  after AI adoption. This is not a conjecture about AI; it is what an
  established model implies.
\end{alertblock}
\end{frame}

\begin{frame}{But Structure Is Contingent}
\begin{itemize}
  \item Identical CT scanners produced \emph{divergent} organizational forms
        in two radiology departments: structure came from technology
        interacting with local context, not from the technology alone.
  \item So the object of study is a \emph{distribution} of responses, not a
        single average effect.
  \item And the stake is the division of \textbf{learning}, not of effort:
        review is how people other than the author learn what is in a system.
\end{itemize}
\end{frame}

\begin{frame}{Two Design Choices That Turned Out to Decide the Answer}
\begin{enumerate}
  \item \textbf{One shared calendar window}, not a window around each
        repository's own adoption date.\\
        Donors now carry a median of \textbf{15} calendar months instead of
        \textbf{5}, so any control can be re-paired with any adopter --- which
        is what makes matching on the baseline outcome possible at all.
  \item \textbf{Closed-unmerged pull requests}, not merged only.\\
        \textbf{26\%} of rows are PRs that never merged. Merged-only panels
        cannot see the review gate, and we show they invert its sign.
\end{enumerate}
\vfill
\begin{block}{Population screen}
  $\geq 3$ distinct human reviewers at baseline. Below that, the top-two
  share is mechanically 1.0 and cannot vary. This is a design requirement,
  not a robustness check.
\end{block}
\end{frame}

\section{Identification}

\begin{frame}{Identification}
\textbf{Allocation (RQ1).} Adopters matched to controls \textbf{on the
baseline outcome} plus volume and size.
\begin{itemize}\small
  \item Baseline top-two gap: $-0.0712 \rightarrow \mathbf{-0.0057}$
  \item \textbf{Zero} pairs at the measure's 1.0 ceiling
  \item 141 matched pairs; two-way FE imputation; 1{,}000 pair bootstraps
\end{itemize}
\vspace{0.5em}
\textbf{The gate (RQ2--RQ3).} Repository $\times$ month fixed effects on the
\emph{opening} cohort.
\begin{itemize}\small
  \item Every comparison is within the same project, same month
  \item No adoption date, no matched control, no bounded share
  \item Clustered by repository; size-adjusted
\end{itemize}
\vspace{0.5em}
\textbf{Both validated by placebo} where the true effect is zero by
construction: $+0.0002$ and $-0.00013$ respectively.
\end{frame}

\section{Results}

\begin{frame}{Result 1: Allocation Does Not Change}
\begin{center}\small
\begin{tabular}{lrrr}
\toprule
Outcome & Baseline & Estimate & $p$\\
\midrule
Top-two review share & 0.820 & $-0.0194$ & 0.088\\
Top-one review share & 0.586 & $-0.0189$ & 0.233\\
Active reviewers / month & 4.83 & $+0.278$ & 0.152\\
Incumbent remains leader & 0.737 & $+0.005$ & 0.892\\
Review events per PR & 1.42 & $+0.077$ & 0.355\\
Median hours to merge & 59.8 & $-9.93$ & 0.622\\
\bottomrule
\end{tabular}
\end{center}
\vfill
\begin{itemize}\small
  \item Nothing survives Holm correction across the family.
  \item Pre-period leads $h=-5\ldots-1$ individually and jointly null
        (joint $p=0.962$): no anticipation, no mis-dated treatment.
  \item The interval \textbf{excludes} the $+0.0243$ a conventional design
        reports.
\end{itemize}
\end{frame}

\begin{frame}{The Theory's Prediction Is Rejected}
\begin{alertblock}{Predicted}
  Review concentrates on the busiest reviewers.
\end{alertblock}
\begin{exampleblock}{Found}
  No detectable change in concentration, reviewer-pool size, incumbent
  continuity, review intensity, or latency.
\end{exampleblock}
\vfill
So where does the adjustment happen? \textbf{Not among reviewers.}
\end{frame}

\begin{frame}{Result 2: It Happens at the Gate}
Share of \emph{opened} pull requests reaching each outcome:
\begin{center}\small
\begin{tabular}{lrrrrr}
\toprule
 & \multicolumn{2}{c}{reviewed} & \multicolumn{2}{c}{no review} & \\
Author & merged & closed & merged & closed & \textbf{merged}\\
\midrule
\textbf{Agent} (1,795) & 44.1\% & 10.9\% & 7.9\% & \textbf{37.0\%} & \textbf{52.0\%}\\
Human (236,386) & 49.9\% & 5.2\% & \textbf{25.0\%} & 19.8\% & \textbf{75.0\%}\\
Dependency bot (48,412) & 42.7\% & 1.9\% & 26.8\% & 28.6\% & 69.5\%\\
\bottomrule
\end{tabular}
\end{center}
\vfill
\begin{itemize}\small
  \item Within repository-month, size-adjusted: agent PRs are
        \textbf{25.7 points} less likely to merge ($t=-8.61$).
  \item Agents are rejected \emph{more} than dependency bots.
\end{itemize}
\end{frame}

\begin{frame}{The Same Observable Means Opposite Things}
\begin{center}\large
\begin{tabular}{lcc}
\toprule
 & $P(\text{merge} \mid$ \textbf{no} review$)$ & $P(\text{merge} \mid$ reviewed$)$\\
\midrule
\textbf{Agent} & \textbf{17.6\%} & 80.2\%\\
Human & \textbf{55.8\%} & 90.6\%\\
Dependency bot & 48.3\% & 95.7\%\\
\bottomrule
\end{tabular}
\end{center}
\vfill
\begin{itemize}
  \item Marginal review rates are \emph{nearly identical}:
        agent 55.0\%, human 55.2\%. Agent code is not globally less reviewed.
  \item What differs is the \textbf{consequence} of going unreviewed.
  \item For an agent PR: abandonment. 82.4\% of unreviewed agent PRs are closed.
  \item For a human PR: self-merge. It goes in anyway, 55.8\% of the time.
\end{itemize}
\end{frame}

\begin{frame}{Result 3: Why --- Authorship Carries Merge Authority}
We collected \texttt{mergedBy}, so this is \textbf{measured, not inferred}:
\begin{center}\small
\begin{tabular}{llrrr}
\toprule
Author & Review status & self-merged & merged by other & self \%\\
\midrule
\textbf{Agent} & reviewed & 0 & 792 & \textbf{0.0\%}\\
\textbf{Agent} & no review & 0 & 142 & \textbf{0.0\%}\\
Human & reviewed & 818 & 1{,}083 & 43.0\%\\
Human & \textbf{no review} & 360 & 257 & \textbf{58.3\%}\\
\bottomrule
\end{tabular}
\end{center}
\vfill
\begin{block}{The asymmetry}
  A human author can merge their own work: 58.3\% of unreviewed human merges
  are self-merges. An agent cannot --- \textbf{0 of 934}. So review is not one
  quality gate among several for machine-authored code. It is the
  \textbf{only route in}.
\end{block}
\end{frame}

\begin{frame}{Result 4: Routing Exists --- It Is Just Invisible}
\begin{itemize}
  \item Conditional on being reviewed, agent PRs are \textbf{+20.3 points}
        more likely to be reviewed by the repository's baseline core
        ($t=+2.16$; core share $+9.2$, $t=+1.87$).
  \item Roster window excluded from estimation, so the core is never defined
        by the PR being scored.
\end{itemize}
\vfill
\begin{block}{This reconciles Results 1 and 4}
  Exception-routing to high-knowledge evaluators \textbf{is} present --- at the
  pull-request level, not the repository-month. It cannot move a monthly share
  because machine-authored work is only \textbf{0.63\%} of opened PRs
  (1{,}795 of 286{,}593). Routing a 0.6\% stream to the core is arithmetically
  invisible in aggregate.
\end{block}
\vfill
\textbf{The mechanism is in place; the volume is not.}
\end{frame}

\begin{frame}{A Falsifiable Forward Prediction}
If agent volume grows and the routing elasticity holds, aggregate
concentration should rise roughly in proportion:
\begin{center}\large
$0.20 \times 10\% \approx \mathbf{+0.02}$ in monthly top-two share
\end{center}
\vfill
\begin{itemize}
  \item That is almost exactly the magnitude a conventional (unidentified)
        analysis reports as \emph{already present}.
  \item Today's null is not evidence of absence. It is evidence that the
        automated stream is still too small to move an aggregate share.
\end{itemize}
\end{frame}

\section{Method}

\begin{frame}{Two Standard Sampling Choices, Two Reversed Conclusions}
\begin{enumerate}
  \item \textbf{Bounded, mean-reverting outcome with unbalanced baselines.}\\
        Control-arm reversion slope $-0.298$ ($t=-5.6$); top-two share sits at
        1.0 in 41\%/53\% of pre-adoption months. A placebo with
        \emph{zero true effect} returns $\mathbf{+0.0111}$ at a $-0.071$
        baseline gap, and $+0.0002$ at ours.
  \item \textbf{Merged-only sampling.}\\
        On merged PRs alone, agent code appears to draw \emph{more} reviewers
        ($+0.225$) with no unreviewed difference ($-0.006$). Observing the
        gate reverses both.
\end{enumerate}
\vfill
\begin{alertblock}{Why standard safeguards miss the first}
  Fixed effects absorb \emph{levels}; reversion makes the expected
  \emph{change} depend on the level. Pre-trend tests examine \emph{slopes}.
  A specification carrying a $-0.071$ gap passes its pre-trend test at $p=0.44$.
\end{alertblock}
\end{frame}

\section{Discussion}

\begin{frame}{What This Says}
\begin{itemize}
  \item Organizations absorb a new kind of producer at the \textbf{boundary},
        not in the \textbf{interior}. They do not redistribute who does the
        complementary work; they change \textbf{what gets admitted}.
  \item Review already functioned as a trust substitute. Human authorship
        supplies that trust directly; machine authorship supplies none, so the
        gate does all the work.
  \item The division of \textbf{learning} is, for now, undisturbed: we find no
        evidence that adoption narrows participation in review.
\end{itemize}
\vfill
\begin{block}{For practice}
  Projects where contributors routinely self-merge have no equivalent path for
  agent-authored work. Nearly half of agent PRs receive no review, and those
  are overwhelmingly abandoned.
\end{block}
\end{frame}

\begin{frame}{Limitations}
\begin{itemize}
  \item \textbf{Agent volume is small}: 1,795 PRs, 0.63\% of the panel, and
        Copilot is 1,234 of them. No agent-identity comparison is supportable.
  \item \textbf{Author classification} is login-pattern based and not yet
        hand-validated.
  \item \textbf{Treatment is first \emph{detected} adoption}, not verified
        first use --- though pre-period leads rule out anticipation.
  \item \textbf{$\geq$40-contributor screen} removes small teams, so results
        describe established projects, not open source generally.
  \item Concentration measures \textbf{allocation}. It is not a measure of
        burden, effort, or strain, and we make no welfare claim.
\end{itemize}
\end{frame}

\begin{frame}{Summary}
\begin{enumerate}
  \item Review allocation among humans \textbf{does not detectably change}
        after AI adoption --- six outcomes, all null, properly identified.
  \item Machine-authored pull requests are \textbf{25.7 points less likely to
        be merged}. The adjustment happens at the gate.
  \item \textbf{Why}: a human author can merge their own work (58.3\% of
        unreviewed merges); an agent never does (0 of 934). Review is the only
        route in for machine-authored code.
  \item Exception-routing to the core \textbf{does} occur, per pull request
        ($+20.3$ points), but at 0.63\% volume it cannot move an aggregate.
  \item Two common sampling choices each \textbf{reverse} one of these
        conclusions.
\end{enumerate}
\end{frame}
